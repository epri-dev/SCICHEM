# Use custom preprocessor to preprocess source if requested

  PREPROCESS source_list_cmake
  COPY source_list_f90 source_list_INC source_list_for
)

if(NOT client_preprocess_only)
  add_executable(METSCI
    ${source_list_f90} 
    ${source_list_for}
  )

  target_include_directories(METSCI
    PRIVATE
      "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/AERMET>"
  )

  # Make sure code has finished preprocessing before use
  set_preprocess_dependency(METSCI)

  # Install
  install(
    TARGETS METSCI
    EXPORT "${TARGETS_EXPORT_NAME}"
    LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}"
    PUBLIC_HEADER DESTINATION "${include_install_dir}"
  )
endif()
